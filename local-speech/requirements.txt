# STT — faster-whisper runs on ctranslate2, NOT torch, so it needs its own
# CUDA libraries even when torch is present. Without these two wheels
# get_stt() falls back to CPU int8 and only says so in one log line.
# ctranslate2 4.x wants cuDNN 9 against CUDA 12.
faster-whisper
nvidia-cublas-cu12
nvidia-cudnn-cu12>=9

# Instrumentation for /metrics. Both were imported but never declared, so the
# Probe's sampling thread died with ModuleNotFoundError on every single request
# — visible only as a traceback in the log and null gpu/cpu/vram fields.
# nvidia-ml-py also replaces shelling out to nvidia-smi, which takes longer to
# start than a GPU stage now takes to finish.
nvidia-ml-py
psutil

# TTS — MMS-TTS (VITS). Thai is the reason: piper has no Thai voice.
# CUDA build, not the CPU wheel: the 20 GB card holds the 8B triage model,
# whisper and VITS at once, so there is nothing left to protect by keeping
# TTS off the GPU.
#
# The +cu124 local-version tag is what makes it a GPU build, and pinning it
# EXACTLY is load-bearing: left unpinned, pip sees candidates on both PyPI and
# the torch index and backtracks through multi-GB wheels for a very long time
# before settling. One candidate means one download.
# 2.6.0 is the only cu124 build for cp313/win_amd64; bump both together.
--extra-index-url https://download.pytorch.org/whl/cu124
torch==2.6.0+cu124
# Pinned to torch's version and pulled from the SAME index. transformers 5.x
# imports torchaudio to load VITS, and installing f5-tts-th drags in whatever
# torchaudio PyPI has latest (2.11.0), which is built against a much newer
# torch — the C++ extension then fails to load with
# "[WinError 127] The specified procedure could not be found" and takes the
# working MMS path down with it. Bump this in lockstep with torch, never alone.
torchaudio==2.6.0+cu124
transformers
